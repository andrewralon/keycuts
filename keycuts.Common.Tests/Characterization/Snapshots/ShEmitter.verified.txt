# ShFormat.EmitLines output. <version> stands for Runner.Version.

## url
|#!/bin/sh|
|# keycuts <version>|
|# <shortcut>/Users/me/Shortcuts/url</shortcut>|
|# <type>url</type>|
|# <target>https://mail.google.com/mail/u/0/#inbox</target>|
|# <description></description>|
|open 'https://mail.google.com/mail/u/0/#inbox'|
|s=$?|
|u="$HOME/Library/Application Support/keycuts/usage"; { mkdir -p "$u" && echo 1 >>"$u/${0##*/}.runs"; } 2>/dev/null|
|exit $s|
